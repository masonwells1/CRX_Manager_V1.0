function readQuoted(sql, start, quote, escapeBackslash = false) {
  let i = start + 1;
  let value = "";
  while (i < sql.length) {
    if (escapeBackslash && sql[i] === "\\" && i + 1 < sql.length) {
      value += sql.slice(i, i + 2);
      i += 2;
      continue;
    }
    if (sql[i] === quote && sql[i + 1] === quote) {
      value += quote;
      i += 2;
      continue;
    }
    if (sql[i] === quote) return { end: i + 1, value };
    value += sql[i++];
  }
  return { end: sql.length, value };
}

function skipTrivia(sql, start) {
  let i = start;
  for (;;) {
    while (/\s/.test(sql[i] || "")) i++;
    if (sql[i] === "-" && sql[i + 1] === "-") {
      const nl = sql.indexOf("\n", i + 2);
      i = nl === -1 ? sql.length : nl + 1;
      continue;
    }
    if (sql[i] === "/" && sql[i + 1] === "*") {
      let depth = 1;
      i += 2;
      while (i < sql.length && depth > 0) {
        if (sql[i] === "/" && sql[i + 1] === "*") { depth++; i += 2; continue; }
        if (sql[i] === "*" && sql[i + 1] === "/") { depth--; i += 2; continue; }
        i++;
      }
      continue;
    }
    return i;
  }
}

function dollarTagAt(sql, index) {
  const match = /^\$[A-Za-z_0-9]*\$/.exec(sql.slice(index));
  return match?.[0] || null;
}

// Split only on terminators outside comments, quotes, and dollar bodies.
function topLevelStatements(sql) {
  const statements = [];
  let start = 0;
  let i = 0;
  while (i < sql.length) {
    if (sql[i] === "'") {
      const escaped = i > 0 && /[eE]/.test(sql[i - 1]) && !/[A-Za-z0-9_$]/.test(sql[i - 2] || "");
      i = readQuoted(sql, i, "'", escaped).end;
      continue;
    }
    if (sql[i] === '"') { i = readQuoted(sql, i, '"').end; continue; }
    if (sql[i] === "-" && sql[i + 1] === "-") {
      const nl = sql.indexOf("\n", i + 2);
      i = nl === -1 ? sql.length : nl + 1;
      continue;
    }
    if (sql[i] === "/" && sql[i + 1] === "*") { i = skipTrivia(sql, i); continue; }
    const tag = sql[i] === "$" ? dollarTagAt(sql, i) : null;
    if (tag) {
      const close = sql.indexOf(tag, i + tag.length);
      i = close === -1 ? sql.length : close + tag.length;
      continue;
    }
    if (sql[i] === ";") {
      const statement = sql.slice(start, i).trim();
      if (statement) statements.push(statement);
      start = ++i;
      continue;
    }
    i++;
  }
  const tail = sql.slice(start).trim();
  if (tail) statements.push(tail);
  return statements;
}

// A keyword-only view: comments and literal values become spaces, while
// identifiers and SQL structure remain available to classifiers.
function keywordView(sql) {
  const src = String(sql || "");
  let out = "";
  let i = 0;
  while (i < src.length) {
    if (src[i] === "'") {
      const escaped = i > 0 && /[eE]/.test(src[i - 1]) && !/[A-Za-z0-9_$]/.test(src[i - 2] || "");
      const quoted = readQuoted(src, i, "'", escaped);
      out += " ".repeat(quoted.end - i);
      i = quoted.end;
      continue;
    }
    if (src[i] === '"') {
      const quoted = readQuoted(src, i, '"');
      out += src.slice(i, quoted.end);
      i = quoted.end;
      continue;
    }
    if (src[i] === "-" && src[i + 1] === "-") {
      const nl = src.indexOf("\n", i + 2);
      const end = nl === -1 ? src.length : nl;
      out += " ".repeat(end - i);
      i = end;
      continue;
    }
    if (src[i] === "/" && src[i + 1] === "*") {
      const end = skipTrivia(src, i);
      out += " ".repeat(end - i);
      i = end;
      continue;
    }
    const tag = src[i] === "$" ? dollarTagAt(src, i) : null;
    if (tag) {
      const close = src.indexOf(tag, i + tag.length);
      const end = close === -1 ? src.length : close + tag.length;
      out += " ".repeat(end - i);
      i = end;
      continue;
    }
    out += src[i++];
  }
  return out;
}

function dmlTargets(statement) {
  const matcher = new RegExp(DML_TARGET_RE.source, "gi");
  return [...keywordView(statement).matchAll(matcher)];
}

function isPgTempTarget(target) {
  return /^(?:pg_temp|pg_temp_\d+)$/i.test(target?.[2] || "");
}
