function readQuoted(sql, start, quote) {
  let i = start + 1;
  let value = "";
  while (i < sql.length) {
    if (sql[i] === quote && sql[i + 1] === quote) {
      value += quote;
      i += 2;
      continue;
    }
    if (sql[i] === quote) return { end: i + 1, value };
    value += sql[i++];
  }
  return { end: sql.length, value };
}

function skipTrivia(sql, start) {
  let i = start;
  for (;;) {
    while (/\s/.test(sql[i] || "")) i++;
    if (sql[i] === "-" && sql[i + 1] === "-") {
      const nl = sql.indexOf("\n", i + 2);
      i = nl === -1 ? sql.length : nl + 1;
      continue;
    }
    if (sql[i] === "/" && sql[i + 1] === "*") {
      let depth = 1;
      i += 2;
      while (i < sql.length && depth > 0) {
        if (sql[i] === "/" && sql[i + 1] === "*") { depth++; i += 2; continue; }
        if (sql[i] === "*" && sql[i + 1] === "/") { depth--; i += 2; continue; }
        i++;
      }
      continue;
    }
    return i;
  }
}

function dollarTagAt(sql, index) {
  const match = /^\$[A-Za-z_0-9]*\$/.exec(sql.slice(index));
  return match?.[0] || null;
}

// Split only on terminators outside comments, quotes, and dollar bodies.
function topLevelStatements(sql) {
  const statements = [];
  let start = 0;
  let i = 0;
  while (i < sql.length) {
    if (sql[i] === "'") { i = readQuoted(sql, i, "'").end; continue; }
    if (sql[i] === '"') { i = readQuoted(sql, i, '"').end; continue; }
    if (sql[i] === "-" && sql[i + 1] === "-") {
      const nl = sql.indexOf("\n", i + 2);
      i = nl === -1 ? sql.length : nl + 1;
      continue;
    }
    if (sql[i] === "/" && sql[i + 1] === "*") { i = skipTrivia(sql, i); continue; }
    const tag = sql[i] === "$" ? dollarTagAt(sql, i) : null;
    if (tag) {
      const close = sql.indexOf(tag, i + tag.length);
      i = close === -1 ? sql.length : close + tag.length;
      continue;
    }
    if (sql[i] === ";") {
      const statement = sql.slice(start, i).trim();
      if (statement) statements.push(statement);
      start = ++i;
      continue;
    }
    i++;
  }
  const tail = sql.slice(start).trim();
  if (tail) statements.push(tail);
  return statements;
}

// A keyword-only view: comments and literal values become spaces, while
// identifiers and SQL structure remain available to classifiers.
function keywordView(sql) {
  const src = String(sql || "");
  let out = "";
  let i = 0;
  while (i < src.length) {
    if (src[i] === "'") {
      const quoted = readQuoted(src, i, "'");
      out += " ".repeat(quoted.end - i);
      i = quoted.end;
      continue;
    }
    if (src[i] === '"') {
      const quoted = readQuoted(src, i, '"');
      out += src.slice(i, quoted.end);
      i = quoted.end;
      continue;
    }
    if (src[i] === "-" && src[i + 1] === "-") {
      const nl = src.indexOf("\n", i + 2);
      const end = nl === -1 ? src.length : nl;
      out += " ".repeat(end - i);
      i = end;
      continue;
    }
    if (src[i] === "/" && src[i + 1] === "*") {
      const end = skipTrivia(src, i);
      out += " ".repeat(end - i);
      i = end;
      continue;
    }
    const tag = src[i] === "$" ? dollarTagAt(src, i) : null;
    if (tag) {
      const close = src.indexOf(tag, i + tag.length);
      const end = close === -1 ? src.length : close + tag.length;
      out += " ".repeat(end - i);
      i = end;
      continue;
    }
    out += src[i++];
  }
  return out;
}

// Recognize the exact executable PL/pgSQL force-abort statement. Text inside
// comments, string literals, quoted identifiers, or nested dollar literals
// cannot qualify.
function hasActualSmokeRaise(body) {
  let i = 0;
  while (i < body.length) {
    if (body[i] === "'") { i = readQuoted(body, i, "'").end; continue; }
    if (body[i] === '"') { i = readQuoted(body, i, '"').end; continue; }
    if (body[i] === "-" && body[i + 1] === "-") {
      const nl = body.indexOf("\n", i + 2);
      i = nl === -1 ? body.length : nl + 1;
      continue;
    }
    if (body[i] === "/" && body[i + 1] === "*") { i = skipTrivia(body, i); continue; }
    const tag = body[i] === "$" ? dollarTagAt(body, i) : null;
    if (tag) {
      const close = body.indexOf(tag, i + tag.length);
      i = close === -1 ? body.length : close + tag.length;
      continue;
    }
    const word = /^[A-Za-z_][A-Za-z0-9_]*/.exec(body.slice(i));
    if (!word) { i++; continue; }
    i += word[0].length;
    if (word[0].toLowerCase() !== "raise") continue;
    i = skipTrivia(body, i);
    const exception = /^[A-Za-z_][A-Za-z0-9_]*/.exec(body.slice(i));
    if (!exception || exception[0].toLowerCase() !== "exception") continue;
    i = skipTrivia(body, i + exception[0].length);
    if ((body[i] === "e" || body[i] === "E") && body[i + 1] === "'") i++;
    if (body[i] !== "'") continue;
    const literal = readQuoted(body, i, "'");
    if (literal.value.toUpperCase() === "SMOKE_PASS_ROLLBACK") return true;
    i = literal.end;
  }
  return false;
}

// A DO abort is rollback proof only when it is the entire submitted batch.
function standaloneDoSmokeAbort(statements) {
  if (statements.length !== 1) return false;
  const statement = stripCommentsQuoteAware(statements[0]);
  const match = /^\s*do(?:\s+language\s+[a-z_][a-z0-9_]*)?\s*(\$[A-Za-z_0-9]*\$)([\s\S]*)\1\s*$/i.exec(statement);
  return Boolean(match && hasActualSmokeRaise(match[2]));
}

function dmlTarget(statement) {
  return DML_TARGET_RE.exec(keywordView(statement));
}

function isPgTempTarget(statement) {
  const target = dmlTarget(statement);
  return Boolean(target && /^(?:pg_temp|pg_temp_\d+)$/i.test(target[2] || ""));
}

function isSyntheticE2ETarget(statement, target) {
  const source = stripCommentsQuoteAware(statement).slice(target.index);
  const verb = target[1].toLowerCase();
  if (verb === "insert into") {
    return /\bvalues\s*\([^;]*'\[E2E\][^']*'/i.test(source);
  }
  if (verb !== "update" && verb !== "delete from") return false;
  const whereIndex = keywordView(source).search(/\bwhere\b/i);
  if (whereIndex === -1) return false;
  const predicate = source.slice(whereIndex);
  if (/\bor\b/i.test(keywordView(predicate))) return false;
  return /(?:=\s*'\[E2E\][^']*'|\b(?:like|ilike)\s*'\[E2E\]%')/i.test(predicate);
}
